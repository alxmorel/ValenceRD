CREATE DATABASE IF NOT EXISTS R_D_Valence;

CREATE TABLE IF NOT EXISTS Produit(
	idProduit INT PRIMARY KEY NOT NULL,
	nomScientifique varchar(255),
	nomGenerique varchar(255)
	);

CREATE TABLE IF NOT EXISTS Etape(
	idEtape INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);

CREATE TABLE IF NOT EXISTS Nomenclature(
	idNomenclature INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);

CREATE TABLE IF NOT EXISTS Version(
	idVersion INT PRIMARY KEY NOT NULL,
	commentaire varchar(255)
	);

CREATE TABLE IF NOT EXISTS Ressource(
	idRessource INT PRIMARY KEY NOT NULL,
	nom varchar(255),
	prenom varchar(100)
	);

CREATE TABLE IF NOT EXISTS Dosage(
	idDosage INT PRIMARY KEY NOT NULL,
	quantite float,
	uniteMesure varchar(50)
	);

CREATE TABLE IF NOT EXISTS Frequence(
	idFrequence INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);

CREATE TABLE IF NOT EXISTS Traitement_Symptome(
	idTraitementSymptome INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);
	
CREATE TABLE IF NOT EXISTS Effet_Secondaire(
	idEffetSecondaire INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);

CREATE TABLE IF NOT EXISTS Phase(
	idPhase INT PRIMARY KEY NOT NULL,
	libelle varchar(255)
	);

	
CREATE TABLE IF NOT EXISTS Travailler(
	idRessource INT NOT NULL,
	idEtape INT NOT NULL,
	PRIMARY KEY(idRessource, idEtape)
	);
	
CREATE TABLE IF NOT EXISTS Manipuler(
	idProduit INT NOT NULL,
	idEtape INT NOT NULL,
	idNomenclature INT NOT NULL,
	quantite float,
	uniteMesure varchar(50),
	PRIMARY KEY(idProduit, idEtape, idNomenclature)
	);

CREATE TABLE IF NOT EXISTS Recette(
	idEtape INT NOT NULL,
	idProduit INT NOT NULL,
	temps float,
	PRIMARY KEY(idEtape, idProduit)
	);

CREATE TABLE IF NOT EXISTS Valider(
	idPhase INT NOT NULL,
	idVersion INT NOT NULL,
	idProduit INT NOT NULL,
	dateInsertion date,
	dateValidation date,
	validation boolean,
	commentaire varchar(5000),
	PRIMARY KEY(idPhase, idVersion, idProduit)
	);
	
CREATE TABLE IF NOT EXISTS Affecter(
	idProduit INT NOT NULL,
	idVersion INT NOT NULL,
	idPhase INT NOT NULL,
	idEssai INT NOT NULL,
	idEffetSecondaire INT NOT NULL,
	idFrequence INT NOT NULL,
	idGravite INT NOT NULL,
	PRIMARY KEY(idProduit, idVersion, idPhase, idEffetSecondaire, idFrequence, idGravite)
	);

CREATE TABLE IF NOT EXISTS Traiter(
	idProduit INT NOT NULL,
	idTraitementSymptome INT NOT NULL,
	PRIMARY KEY(idProduit, idTraitementSymptome)
	);

CREATE TABLE IF NOT EXISTS Administrer(
	idProduit INT NOT NULL,
	idDosage INT NOT NULL,
	idFrequence INT NOT NULL,
	PRIMARY KEY(idProduit, idDosage, idFrequence)
	);
